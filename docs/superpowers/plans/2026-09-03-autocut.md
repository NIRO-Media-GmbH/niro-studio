# NIRO AutoCut — Umsetzungsplan (Stufe 1–3)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Aus einem Cutter-Schnittplan eine Rohschnitt-Timeline in DaVinci Resolve bauen (V1 FX3, V2 a7IV synchron, A1/A2, Platzhalter, Marker), alle B-Roll-Clips des Drehs per Claude-Vision indexieren und V3 passend füllen.

**Architecture:** Neues, eigenständiges Werkzeug `tools/autocut/` (Python 3.12, eigene venv). Claude entscheidet in der Session (Cutlist, B-Roll-Zuordnung); Code prüft hart und baut per Resolve-Scripting-API. Bestehende Tools werden nur importiert (`niro_transcribe` per `.pth`), nie verändert. Alle Arbeitsdateien liegen in `_intern/autocut/` der Charge, Berichte in `Ergebnisse/Rohschnitt/`.

**Tech Stack:** Python 3.12, numpy/scipy (Sync), rapidfuzz (Alignment), Pillow (Kontaktbögen), anthropic ≥ 1.3 (Vision, Structured Outputs), ffmpeg/ffprobe 8, DaVinci Resolve Studio 21 Scripting API, pytest, pyyaml, python-dotenv.

Spec: `docs/superpowers/specs/2026-09-03-autocut-design.md`

## Global Constraints

- **Nichts außerhalb von `tools/autocut/` ändern**, einzige Ausnahme: eine neue Zeile in der Trigger-Tabelle von `CLAUDE.md` (Task 12).
- **NAS nur lesen.** Kein Skript schreibt unter `/Volumes/`. Arbeitsdateien nur unter `<Charge>/_intern/autocut/`, Berichte nur unter `<Charge>/Ergebnisse/Rohschnitt/`, Protokoll nur anhängen.
- **Resolve:** nur neue Bin/Timeline/Media-Pool-Einträge anlegen; bestehende Timelines nie verändern; löschen nur Timelines, die derselbe Lauf erzeugt hat (Probe).
- venv: `tools/autocut/venv/bin/python` (Python 3.12.5). `niro_transcribe` ist über `venv/lib/python3.12/site-packages/niro_transcribe.pth` importierbar — **nicht** per pip installieren.

- **Keine git-Commits** in diesem Durchgang (User hat nicht darum gebeten; Repo enthält fremde, uncommittete Arbeit). Stand bleibt im Arbeitsbaum.

```python
# src/niro_autocut/charge.py
"""Charge (projects/<Kunde>/<Projekt>/<Charge>) öffnen, Config laden, Schreibbereiche absichern."""
from __future__ import annotations

import datetime as _dt
import json
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

import yaml

TOOL_ROOT = Path(__file__).resolve().parents[2]
DEFAULTS_FILE = TOOL_ROOT / "defaults.yaml"


class AutoCutError(Exception):
    """Fehler mit deutscher, handlungsleitender Meldung."""


def _deep_merge(base: dict, override: dict) -> dict:
    out = dict(base)
    for k, v in override.items():
        if isinstance(v, dict) and isinstance(out.get(k), dict):
            out[k] = _deep_merge(out[k], v)
        else:
            out[k] = v
    return out


def load_config(charge_root: str | Path) -> dict:
    cfg = yaml.safe_load(DEFAULTS_FILE.read_text(encoding="utf-8")) or {}
    local = Path(charge_root) / "_intern" / "autocut" / "config.yaml"
    if local.exists():
        cfg = _deep_merge(cfg, yaml.safe_load(local.read_text(encoding="utf-8")) or {})
    return cfg


@dataclass
class Charge:
    root: Path
    intern: Path
    autocut: Path
    work: Path
    ergebnisse: Path
    plaene: Path
    protokoll: Path
    config: dict = field(default_factory=dict)

    @classmethod
    def open(cls, root: str | Path) -> "Charge":
        root = Path(root).expanduser().resolve()
        plaene = root / "Ergebnisse" / "O-Ton-Pläne"
        if not plaene.is_dir():
            raise AutoCutError(
                f"Keine Charge gefunden: {root}\nErwartet wird der Ordner Ergebnisse/O-Ton-Pläne "
                f"mit einem Cutter-Plan (video-N-*.md). Erst den Schnittplan-Workflow ausführen.")
        intern = root / "_intern"
        if not (intern / "utterances.json").exists():
            raise AutoCutError(f"{intern / 'utterances.json'} fehlt — bitte zuerst "
                               f"tools/transcribe/scripts/build_utterances.py für diese Charge ausführen.")
        ch = cls(root=root, intern=intern, autocut=intern / "autocut", work=intern / "autocut" / "work",
                 ergebnisse=root / "Ergebnisse" / "Rohschnitt", plaene=plaene, protokoll=root / "Protokoll.md",
                 config=load_config(root))
        for d in (ch.autocut, ch.work, ch.ergebnisse):
            d.mkdir(parents=True, exist_ok=True)
        return ch

    # --- Schreibschutz -------------------------------------------------
    def assert_writable(self, path: str | Path) -> None:
        p = Path(path).resolve()
        allowed = (self.autocut.resolve(), self.ergebnisse.resolve())
        if p == self.protokoll.resolve():
            return
        if not any(str(p).startswith(str(a) + "/") or p == a for a in allowed):
            raise AutoCutError(f"Schreiben verweigert: {p}\nAutoCut schreibt nur unter "
                               f"{self.autocut} und {self.ergebnisse} sowie ans Protokoll.")

    # --- Eingaben ------------------------------------------------------
    def plan_files(self) -> list[Path]:
        return sorted(p for p in self.plaene.glob("video-*.md") if p.is_file())

    def resolve_plan(self, name: str | None) -> Path:
        plans = self.plan_files()
        if name:
            for p in plans:
                if p.name == name or p.stem == name:
                    return p
            raise AutoCutError(f"Plan '{name}' nicht gefunden in {self.plaene}. Vorhanden: "
                               + ", ".join(p.name for p in plans))
        if len(plans) == 1:
            return plans[0]
        if not plans:
            raise AutoCutError(f"Kein Cutter-Plan (video-*.md) in {self.plaene}.")
        raise AutoCutError("Mehrere Pläne vorhanden, bitte mit --video wählen: "
                           + ", ".join(p.name for p in plans))

    def load_index(self) -> list[dict]:
        p = self.intern / "transcripts_index.json"
        if not p.exists():
            raise AutoCutError(f"{p} fehlt — Transkript-Index der Charge nicht vorhanden.")
        return json.loads(p.read_text(encoding="utf-8"))

    def load_utterances(self) -> list[dict]:
        return json.loads((self.intern / "utterances.json").read_text(encoding="utf-8"))

    def cache_transcript(self, fingerprint: str) -> dict | None:
        p = self.intern / "cache" / f"{fingerprint}.scribe.json"
        return json.loads(p.read_text(encoding="utf-8")) if p.exists() else None

    # --- Arbeitsdateien -------------------------------------------------
    def read_json(self, name: str) -> Any | None:
        p = self.autocut / name
        return json.loads(p.read_text(encoding="utf-8")) if p.exists() else None

    def write_json(self, name: str, data: Any) -> Path:
        p = self.autocut / name
        self.assert_writable(p)
        p.write_text(json.dumps(data, ensure_ascii=False, indent=1), encoding="utf-8")
        return p


def append_protokoll(charge: Charge, titel: str, zeilen: list[str]) -> None:
    charge.assert_writable(charge.protokoll)
    stamp = _dt.datetime.now().strftime("%Y-%m-%d %H:%M")
    block = [f"\n## {stamp} — AutoCut: {titel}\n"] + [f"- {z}" for z in zeilen] + [""]
    existing = charge.protokoll.read_text(encoding="utf-8") if charge.protokoll.exists() else "# Protokoll\n"
    charge.protokoll.write_text(existing.rstrip("\n") + "\n" + "\n".join(block), encoding="utf-8")
```

```python
"""Legt tools/autocut/.env mit ANTHROPIC_API_KEY an (Quelle: macOS-Schlüsselbund, Dienst niro_autocut)."""
from __future__ import annotations
import subprocess, sys
from pathlib import Path

ENV = Path(__file__).resolve().parents[1] / ".env"

def main() -> None:
    if ENV.exists() and "ANTHROPIC_API_KEY=" in ENV.read_text():
        print(f".env vorhanden: {ENV}"); return
    r = subprocess.run(["security", "find-generic-password", "-s", "niro_autocut", "-a", "anthropic", "-w"],
                       capture_output=True, text=True)
    key = r.stdout.strip()
    if r.returncode != 0 or not key.startswith("sk-ant-"):
        sys.exit("Kein Anthropic-Key im Schlüsselbund (Dienst niro_autocut, Konto anthropic). "
                 "Bitte ANTHROPIC_API_KEY=... von Hand in tools/autocut/.env eintragen.")
    ENV.write_text(f"# NIRO AutoCut — nicht versionieren\nANTHROPIC_API_KEY={key}\n", encoding="utf-8")
    ENV.chmod(0o600)
    print(f".env geschrieben: {ENV} (Key endet auf …{key[-4:]})")

if __name__ == "__main__":
    main()
```

```python
PROXY_DIR = "Proxy"
PROXY_EXTS = (".mov", ".mp4", ".MOV", ".MP4")


@dataclass
class MediaInfo:
    path: str
    duration_s: float
    fps: float
    width: int
    height: int
    rotation: int
    nb_frames: int
    timecode: str | None
    has_audio: bool
    sample_rate: int
    channels: int

    def to_dict(self) -> dict:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: dict) -> "MediaInfo":
        return cls(**{k: d[k] for k in cls.__dataclass_fields__})


def _which(name: str) -> str:
    p = shutil.which(name)
    if not p:
        raise AutoCutError(f"{name} nicht gefunden (brew install ffmpeg).")
    return p


def _fps(s: str) -> float:
    if "/" in s:
        n, d = s.split("/")
        return round(int(n) / int(d), 3) if int(d) else 0.0
    return float(s or 0)


def _parse_probe(path: str, data: dict) -> MediaInfo:
    v = next((s for s in data.get("streams", []) if s.get("codec_type") == "video"), None)
    if v is None:
        raise AutoCutError(f"Kein Videostream in {path}")
    a = next((s for s in data.get("streams", []) if s.get("codec_type") == "audio"), None)
    fps = _fps(v.get("avg_frame_rate") or v.get("r_frame_rate") or "0/1") or _fps(v.get("r_frame_rate", "0/1"))
    duration = float(data.get("format", {}).get("duration") or v.get("duration") or 0.0)
    rotation = 0
    for sd in v.get("side_data_list", []) or []:
        if "rotation" in sd:
            rotation = abs(int(round(float(sd["rotation"])))) % 360
    if not rotation and v.get("tags", {}).get("rotate"):
        rotation = abs(int(v["tags"]["rotate"])) % 360
    nb = int(v["nb_frames"]) if str(v.get("nb_frames", "")).isdigit() else round(duration * fps)
    tc = None
    for s in data.get("streams", []):
        t = (s.get("tags") or {}).get("timecode")
        if t:
            tc = t
            break
    if tc is None:
        tc = (data.get("format", {}).get("tags") or {}).get("timecode")
    return MediaInfo(path=str(path), duration_s=duration, fps=fps, width=int(v.get("width", 0)),
                     height=int(v.get("height", 0)), rotation=rotation, nb_frames=nb, timecode=tc,
                     has_audio=a is not None, sample_rate=int((a or {}).get("sample_rate", 0) or 0),
                     channels=int((a or {}).get("channels", 0) or 0))


def ffprobe(path: str | Path) -> MediaInfo:
    p = Path(path)
    if not p.is_file():
        raise AutoCutError(f"Datei nicht gefunden: {p}\nIst das NAS gemountet?")
    cmd = [_which("ffprobe"), "-v", "error", "-print_format", "json", "-show_format", "-show_streams", str(p)]
    r = subprocess.run(cmd, capture_output=True, text=True)
    if r.returncode != 0:
        raise AutoCutError(f"ffprobe fehlgeschlagen für {p.name}: {r.stderr[-300:]}")
    return _parse_probe(str(p), json.loads(r.stdout))


def proxy_for(path: str | Path) -> Path | None:
    p = Path(path)
    for ext in PROXY_EXTS:
        cand = p.parent / PROXY_DIR / (p.stem + ext)
        if cand.is_file():
            return cand
    return None


def fingerprint(path: str | Path) -> str:
    return clip_fingerprint(path)


def is_portrait(info: MediaInfo) -> bool:
    if info.rotation in (90, 270):
        return info.width >= info.height
    return info.height > info.width


def timeline_format(info: MediaInfo) -> dict:
    w, h = (info.width, info.height)
    if info.rotation in (90, 270):
        w, h = h, w
    return {"fps": info.fps, "width": w, "height": h, "orientation": "9:16" if h > w else "16:9"}


def check_proxy_match(orig: MediaInfo, proxy: MediaInfo, tol_frames: int = 1) -> list[str]:
    probs: list[str] = []
    if abs(orig.fps - proxy.fps) > 0.01:
        probs.append(f"Bildrate weicht ab: Original {orig.fps} / Proxy {proxy.fps}")
    if abs(orig.nb_frames - proxy.nb_frames) > tol_frames + 1:
        probs.append(f"Frames weichen ab: Original {orig.nb_frames} / Proxy {proxy.nb_frames}")
    return probs


def seconds_to_frames(s: float, fps: float) -> int:
    return int(round(s * fps))


def frames_to_seconds(f: int, fps: float) -> float:
    return round(f / fps, 3)


def extract_audio_16k(src: str | Path, out_dir: str | Path, sr: int = 16000) -> Path:
    """Mono-WAV aus dem ORIGINAL (lesend), gecacht per Fingerprint im out_dir."""
    src = Path(src)
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    target = out_dir / f"{src.stem}.{fingerprint(src)[:12]}.wav"
    if target.exists() and target.stat().st_size > 1000:
        return target
    cmd = [_which("ffmpeg"), "-hide_banner", "-loglevel", "error", "-y", "-i", str(src), "-vn",
           "-ac", "1", "-ar", str(sr), "-c:a", "pcm_s16le", str(target)]
    r = subprocess.run(cmd, capture_output=True, text=True)
    if r.returncode != 0 or not target.exists():
        raise AutoCutError(f"Audio-Extraktion fehlgeschlagen für {src.name}: {r.stderr[-300:]}")
    return target
```

```python
# src/niro_autocut/plan.py
"""Cutter-Plan (Markdown-Tabelle des Schnittplan-Workflows) lesen — nur lesend."""
from __future__ import annotations

import re
from dataclasses import dataclass, field, asdict
from pathlib import Path

from .charge import AutoCutError

STEM_RE = re.compile(r"\b((?:FX3|a7MK4|DJI|C)_?[0-9]{3,}(?:_[0-9]+)?)\b")
TC_RE = re.compile(r"(\d{1,2}:\d{2}(?::\d{2})?)\s*(?:[–\-]\s*(\d{1,2}:\d{2}(?::\d{2})?))?")
DAUER_RE = re.compile(r"~\s*(\d+(?:[.,]\d+)?)\s*s\b")
ZIEL_RE = re.compile(r"Ziell[äa]nge[^0-9]{0,6}(?:≈\s*)?(\d{1,2}):(\d{2})|Ziell[äa]nge[^0-9]{0,6}(\d{2,3})(?:\s*[–\-]\s*(\d{2,3}))?\s*s")


@dataclass
class PlanRow:
    nr: str
    szene: str
    oton: str
    quelle: str
    bild: str
    sound: str
    caption: str
    kommentar: str
    typ: str
    plan_dauer_s: float | None
    tc_hints: list[dict] = field(default_factory=list)

    def to_dict(self) -> dict:
        return asdict(self)


@dataclass
class Plan:
    file: str
    titel: str
    ziel_laenge_s: float | None
    format_hint: str | None
    rows: list[PlanRow]
    verboten_text: str
    material_text: str

    def to_dict(self) -> dict:
        d = asdict(self)
        return d


def parse_mmss(s: str) -> float:
    parts = [int(x) for x in s.strip().split(":")]
    if len(parts) == 2:
        return parts[0] * 60 + parts[1]
    if len(parts) == 3:
        return parts[0] * 3600 + parts[1] * 60 + parts[2]
    raise ValueError(s)


def parse_plan_dauer(kommentar: str) -> float | None:
    m = DAUER_RE.search(kommentar or "")
    return float(m.group(1).replace(",", ".")) if m else None


def parse_ziel_laenge(text: str) -> float | None:
    m = ZIEL_RE.search(text or "")
    if not m:
        return None
    if m.group(1):
        return int(m.group(1)) * 60 + int(m.group(2))
    lo, hi = m.group(3), m.group(4)
    return float(hi or lo)


def parse_tc_hints(text: str) -> list[dict]:
    """Findet Datei-Stems und mm:ss(–mm:ss)-Angaben; jede Zeitangabe gehört zum zuletzt genannten Stem."""
    hints: list[dict] = []
    stem = None
    tokens = re.split(r"(\s+|·|\+|/)", text or "")
    for tok in tokens:
        ms = STEM_RE.search(tok)
        if ms:
            stem = ms.group(1)
            continue
        mt = TC_RE.fullmatch(tok.strip()) if tok.strip() else None
        if mt and stem:
            von = parse_mmss(mt.group(1))
            bis = parse_mmss(mt.group(2)) if mt.group(2) else None
            hints.append({"stem": stem, "von_s": von, "bis_s": bis})
    # Fallback für zusammenhängende Schreibweise "FX3_0543 · 12:03–12:34"
    if not hints:
        stems = STEM_RE.findall(text or "")
        for mt in TC_RE.finditer(text or ""):
            if stems:
                bis = parse_mmss(mt.group(2)) if mt.group(2) else None
                hints.append({"stem": stems[-1], "von_s": parse_mmss(mt.group(1)), "bis_s": bis})
    return hints


def classify_row(oton: str, quelle: str, bild: str) -> str:
    o = (oton or "").strip()
    q = (quelle or "").strip()
    if o.upper().startswith("VO") or q.upper().startswith("VO"):
        return "vo"
    if o in ("", "—", "-", "–") or o.startswith("—"):
        b = (bild or "").lower()
        if "grafik" in b or "endcard" in b or "motion" in b:
            return "grafik"
        return "bild"
    return "oton"


def _split_row(line: str) -> list[str]:
    cells = [c.strip() for c in line.strip().strip("|").split("|")]
    return cells


def parse_plan(path: str | Path) -> Plan:
    p = Path(path)
    text = p.read_text(encoding="utf-8")
    titel = next((l.lstrip("# ").strip() for l in text.splitlines() if l.startswith("# ")), p.stem)
    ziel = parse_ziel_laenge(text)
    fmt = "9:16" if re.search(r"\b9:16\b", text) and not re.search(r"\b16:9\b", text) else ("16:9" if "16:9" in text else None)
    verboten = ""
    mv = re.search(r"\*\*Verboten:\*\*(.+?)(?:\n\n|\n\*\*|\n## )", text, re.S)
    if mv:
        verboten = mv.group(1).strip()
    material = ""
    mm = re.search(r"## Material\n(.+?)\n## ", text, re.S)
    if mm:
        material = mm.group(1).strip()
    rows: list[PlanRow] = []
    header: list[str] | None = None
    for line in text.splitlines():
        if not line.strip().startswith("|"):
            continue
        cells = _split_row(line)
        if header is None:
            if cells and cells[0] == "#":
                header = cells
            continue
        if all(set(c) <= set("-: ") for c in cells):
            continue
        if len(cells) < 8:
            continue
        nr, szene, oton, quelle, bild, sound, caption, kommentar = cells[:8]
        if not nr:
            continue
        typ = classify_row(oton, quelle, bild)
        rows.append(PlanRow(nr=nr, szene=szene, oton=oton, quelle=quelle, bild=bild, sound=sound,
                            caption=caption, kommentar=kommentar, typ=typ,
                            plan_dauer_s=parse_plan_dauer(kommentar), tc_hints=parse_tc_hints(quelle)))
    if not rows:
        raise AutoCutError(f"Keine Ablauf-Tabelle im Plan gefunden: {p}")
    return Plan(file=str(p), titel=titel, ziel_laenge_s=ziel, format_hint=fmt, rows=rows,
                verboten_text=verboten, material_text=material)
```

```python
# src/niro_autocut/align.py  (URSPRÜNGLICHER ENTWURF — durch die Hülle um quote_align ersetzen, API beibehalten)
"""Wörtliche Zitate gegen Wort-Zeitstempel (Scribe) ausrichten."""
from __future__ import annotations

import re
import unicodedata
from dataclasses import dataclass

from rapidfuzz import fuzz

ELISION_RE = re.compile(r"\[\s*(?:…|\.\.\.)\s*\]|(?<!\w)…(?!\w)|(?<!\w)\.\.\.(?!\w)")
QUOTES = "„“”\"‚‘’'«»"
NUM_WORDS = {"hundert": "100", "einhundert": "100", "zehn": "10", "zwölf": "12", "zwanzig": "20",
             "dreißig": "30", "vierzig": "40", "fünfzig": "50", "sechzig": "60", "siebzig": "70",
             "achtzig": "80", "neunzig": "90", "vierundzwanzig": "24", "prozent": "%"}


def normalize(tok: str) -> str:
    t = unicodedata.normalize("NFC", tok or "").lower().strip()
    t = t.replace("ß", "ss")
    t = "".join(ch for ch in t if ch.isalnum() or ch in "%")
    return NUM_WORDS.get(t, t)


def split_fragments(quote: str) -> list[str]:
    q = (quote or "").strip().strip(QUOTES).strip()
    parts = [p.strip().strip(QUOTES).strip() for p in ELISION_RE.split(q)]
    return [p for p in parts if p]


@dataclass
class Match:
    start_idx: int
    end_idx: int
    start_s: float
    end_s: float
    score: float
    text: str
    speaker: str | None


def _norm_words(words: list[dict]) -> list[str]:
    return [normalize(w.get("text", "")) for w in words]


def find_span(words: list[dict], fragment: str, near_s: float | None = None, min_score: float = 0.6) -> list[Match]:
    frag = [t for t in (normalize(x) for x in fragment.split()) if t]
    if not frag:
        return []
    nw = _norm_words(words)
    n = len(frag)
    target = " ".join(frag)
    cands: list[Match] = []
    for length in sorted({max(1, n - 2), max(1, n - 1), n, n + 1, n + 2}):
        for i in range(0, len(nw) - length + 1):
            window = nw[i:i + length]
            if not window[0] or not window[-1]:
                continue
            if window[0] != frag[0] and fuzz.ratio(window[0], frag[0]) < 60:
                continue
            score = fuzz.ratio(" ".join(window), target) / 100.0
            if score >= min_score:
                w0, w1 = words[i], words[i + length - 1]
                cands.append(Match(i, i + length - 1, float(w0["start"]), float(w1["end"]), score,
                                   " ".join(w.get("text", "") for w in words[i:i + length]), w0.get("speaker")))
    def key(m: Match):
        dist = abs(m.start_s - near_s) if near_s is not None else 0.0
        return (-round(m.score, 2), dist, m.start_s)
    cands.sort(key=key)
    # Überlappende Kandidaten ausdünnen (bester pro Startbereich)
    out: list[Match] = []
    for m in cands:
        if all(abs(m.start_s - o.start_s) > 0.5 for o in out):
            out.append(m)
    return out


def find_quote(words: list[dict], quote: str, near_s: float | None = None, min_score: float = 0.8,
               max_gap_s: float = 180.0) -> Match | None:
    frags = split_fragments(quote)
    if not frags:
        return None
    firsts = find_span(words, frags[0], near_s=near_s, min_score=min_score)
    if not firsts:
        return None
    if len(frags) == 1:
        return firsts[0]
    for first in firsts[:5]:
        lasts = [m for m in find_span(words, frags[-1], near_s=first.start_s, min_score=min_score)
                 if m.start_s >= first.end_s - 0.01 and m.start_s - first.end_s <= max_gap_s]
        if lasts:
            last = min(lasts, key=lambda m: (m.start_s - first.end_s))
            score = min(first.score, last.score)
            return Match(first.start_idx, last.end_idx, first.start_s, last.end_s, score,
                         " ".join(w.get("text", "") for w in words[first.start_idx:last.end_idx + 1]), first.speaker)
    return None


def words_between(words: list[dict], start_s: float, end_s: float) -> list[dict]:
    return [w for w in words if start_s <= (float(w["start"]) + float(w["end"])) / 2 <= end_s]
```

```python
# src/niro_autocut/cutlist.py
"""Cutlist: was Claude aus dem Plan macht, und die harte Prüfung davor dem Bau."""
from __future__ import annotations

import hashlib
import json
from dataclasses import dataclass, field, asdict
from pathlib import Path

from .align import find_quote, words_between
from .charge import AutoCutError

TYPEN = ("oton", "vo", "bild", "grafik")


@dataclass
class Cut:
    in_s: float
    out_s: float
    text: str = ""
    hart_in: bool = False
    hart_out: bool = False


@dataclass
class Beat:
    nr: str
    szene: str
    typ: str
    person: str | None = None
    rolle: str | None = None
    clip: str | None = None
    cuts: list[Cut] = field(default_factory=list)
    pause_after_s: float | None = None
    platzhalter_s: float | None = None
    text: str | None = None
    bild_hinweis: str = ""
    kommentar: str = ""
    caption: str | None = None
    plan_dauer_s: float | None = None
    sound: str = ""

    def dauer_s(self) -> float:
        if self.typ == "oton":
            return sum(c.out_s - c.in_s for c in self.cuts)
        return float(self.platzhalter_s or 0.0)


@dataclass
class Sperre:
    clip: str
    von_s: float
    bis_s: float
    grund: str = ""


@dataclass
class Cutlist:
    video: str
    ziel_laenge_s: float | None
    fps: float
    format: str
    pause_s: float
    beats: list[Beat]
    sperren: list[Sperre] = field(default_factory=list)

    def to_dict(self) -> dict:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: dict) -> "Cutlist":
        beats = [Beat(**{**b, "cuts": [Cut(**c) for c in b.get("cuts", [])]}) for b in d.get("beats", [])]
        return cls(video=d["video"], ziel_laenge_s=d.get("ziel_laenge_s"), fps=float(d["fps"]),
                   format=d.get("format", "16:9"), pause_s=float(d.get("pause_s", 1.0)), beats=beats,
                   sperren=[Sperre(**s) for s in d.get("sperren", [])])

    def save(self, path: str | Path) -> None:
        Path(path).write_text(json.dumps(self.to_dict(), ensure_ascii=False, indent=1), encoding="utf-8")

    @classmethod
    def load(cls, path: str | Path) -> "Cutlist":
        return cls.from_dict(json.loads(Path(path).read_text(encoding="utf-8")))


@dataclass
class VerifyResult:
    errors: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)

    @property
    def ok(self) -> bool:
        return not self.errors


def cutlist_hash(path: str | Path) -> str:
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def total_length_s(cl: Cutlist, cfg: dict) -> float:
    total = 0.0
    for i, b in enumerate(cl.beats):
        total += b.dauer_s()
        if i < len(cl.beats) - 1:
            total += b.pause_after_s if b.pause_after_s is not None else cl.pause_s
    return total


def _overlaps(a0: float, a1: float, b0: float, b1: float) -> bool:
    return a0 < b1 and b0 < a1


def verify_cutlist(cl: Cutlist, charge, words_by_clip: dict[str, list[dict]], durations: dict[str, float],
                   cfg: dict, sync: dict | None = None) -> VerifyResult:
    r = VerifyResult()
    fps = cl.fps
    h_in = cfg["handle_in_frames"] / fps
    h_out = cfg["handle_out_frames"] / fps
    lo, hi = cfg["duration_tolerance"]
    for b in cl.beats:
        tag = f"Beat #{b.nr} ({b.szene})"
        if b.typ not in TYPEN:
            r.errors.append(f"{tag}: unbekannter Typ '{b.typ}'")
            continue
        if b.typ != "oton":
            if not b.platzhalter_s or b.platzhalter_s <= 0:
                r.errors.append(f"{tag}: platzhalter_s fehlt (Plan-Schätzung „~x s" übernehmen)")
            continue
        if not b.clip or not b.cuts:
            r.errors.append(f"{tag}: clip oder cuts fehlen"); continue
        words = words_by_clip.get(b.clip)
        if words is None:
            r.errors.append(f"{tag}: Clip {b.clip} hat kein Transkript oder ist nicht in der Charge"); continue
        dur = durations.get(b.clip)
        for c in b.cuts:
            if c.in_s < 0 or c.out_s <= c.in_s:
                r.errors.append(f"{tag}: ungültiges Intervall {c.in_s}–{c.out_s}"); continue
            if dur is not None and c.out_s > dur + 0.05:
                r.errors.append(f"{tag}: out {c.out_s}s liegt hinter dem Clip-Ende {dur}s")
            inside = words_between(words, c.in_s, c.out_s)
            if not inside:
                r.errors.append(f"{tag}: kein Transkript-Wort in {c.in_s}–{c.out_s}"); continue
            if c.text:
                m = find_quote(inside, c.text, min_score=cfg["align_min_score"])
                if m is None:
                    got = " ".join(w["text"] for w in inside)[:120]
                    r.errors.append(f"{tag}: Text stimmt nicht mit dem Transkript im Intervall überein. Dort steht: „{got}…\"")
            a0 = c.in_s - (0 if c.hart_in else h_in)
            a1 = c.out_s + (0 if c.hart_out else h_out)
            for s in cl.sperren:
                if s.clip == b.clip and _overlaps(a0, a1, s.von_s, s.bis_s):
                    r.errors.append(f"{tag}: überschneidet Sperre {s.von_s}–{s.bis_s}s ({s.grund})")
        if b.plan_dauer_s:
            d = b.dauer_s()
            if not (lo * b.plan_dauer_s <= d <= hi * b.plan_dauer_s):
                r.warnings.append(f"{tag}: Dauer {d:.1f}s weicht stark von der Plan-Schätzung ~{b.plan_dauer_s:g}s ab")
        if sync is not None:
            paare = [p for p in sync.get("paare", []) if p["ref"] == b.clip and p["ok"]]
            for c in b.cuts:
                covered = any(p["overlap_ref"][0] <= c.in_s and c.out_s <= p["overlap_ref"][1] for p in paare)
                if not covered:
                    r.warnings.append(f"{tag}: keine a7-Abdeckung für {c.in_s:.1f}–{c.out_s:.1f}s (V2 bleibt leer)")
    if cl.ziel_laenge_s:
        total = total_length_s(cl, cfg)
        if abs(total - cl.ziel_laenge_s) / cl.ziel_laenge_s > cfg["target_length_warn"]:
            r.warnings.append(f"Gesamtlänge {total:.0f}s weicht von der Ziellänge {cl.ziel_laenge_s:.0f}s um mehr als "
                              f"{int(cfg['target_length_warn'] * 100)} % ab")
    return r
```

```python
# src/niro_autocut/sync.py
"""Waveform-Sync zweier Kameras: Onset-Hüllkurven + FFT-Kreuzkorrelation. Konvention: other_time = ref_time + offset_s."""
from __future__ import annotations

from dataclasses import dataclass, asdict
from pathlib import Path

import numpy as np
from scipy.io import wavfile
from scipy.signal import correlate, correlation_lags

from .charge import AutoCutError
from .media import extract_audio_16k


def read_wav_mono(path: str | Path) -> tuple[np.ndarray, int]:
    sr, x = wavfile.read(str(path))
    if x.ndim > 1:
        x = x.mean(axis=1)
    x = x.astype(np.float32)
    if x.size and np.abs(x).max() > 1.5:
        x = x / 32768.0
    return x, int(sr)


def onset_envelope(x: np.ndarray, sr: int, hop_s: float = 0.005, win_s: float = 0.02) -> tuple[np.ndarray, float]:
    h = max(1, int(sr * hop_s)); w = max(h, int(sr * win_s))
    n = (len(x) - w) // h
    if n <= 10:
        raise AutoCutError("Audio zu kurz für den Sync (< 1 s).")
    idx = np.arange(w)[None, :] + h * np.arange(n)[:, None]
    e = np.sqrt((x[idx] ** 2).mean(axis=1) + 1e-10)
    db = 20 * np.log10(e + 1e-6)
    db = db - np.median(db)
    on = np.maximum(np.diff(db, prepend=db[0]), 0.0)
    on = on - on.mean()
    return on.astype(np.float32), 1.0 / hop_s


def estimate_offset(ref_env: np.ndarray, other_env: np.ndarray, fs: float, exclusion_s: float = 1.0) -> tuple[float, float]:
    c = correlate(ref_env, other_env, mode="full", method="fft")
    lags = correlation_lags(len(ref_env), len(other_env), mode="full")
    k = int(np.argmax(c))
    lag_s = lags[k] / fs
    excl = max(1, int(fs * exclusion_s))
    c2 = c.copy(); c2[max(0, k - excl):k + excl] = -np.inf
    second = float(np.max(c2)) if np.isfinite(c2).any() else 1e-9
    conf = float(c[k] / max(second, 1e-9)) if c[k] > 0 else 0.0
    return float(-lag_s), conf


@dataclass
class PairSync:
    ref: str
    other: str
    offset_s: float
    offset_frames: int
    confidence: float
    overlap_ref: list[float]
    drift_frames: float
    ok: bool
    note: str = ""

    def to_dict(self) -> dict:
        return asdict(self)


def sync_pair(ref_wav: str | Path, other_wav: str | Path, fps: float, cfg: dict) -> PairSync:
    ref, sr = read_wav_mono(ref_wav); oth, sr2 = read_wav_mono(other_wav)
    if sr != sr2:
        raise AutoCutError("Abtastraten der Sync-WAVs unterscheiden sich.")
    hop, win, excl = cfg["hop_s"], cfg["win_s"], cfg["exclusion_s"]
    e_ref, fs = onset_envelope(ref, sr, hop, win); e_oth, _ = onset_envelope(oth, sr, hop, win)
    off, conf = estimate_offset(e_ref, e_oth, fs, excl)
    ref_dur, oth_dur = len(ref) / sr, len(oth) / sr
    ov0, ov1 = max(0.0, -off), min(ref_dur, oth_dur - off)
    note = ""
    ok = conf >= cfg["min_confidence"] and ov1 - ov0 >= 5.0
    drift = 0.0
    if ok:
        mid = (ov0 + ov1) / 2
        ests = []
        for a, b in ((ov0, mid), (mid, ov1)):
            i0, i1 = int(a * fs), int(b * fs)
            if i1 - i0 < int(5 * fs):
                continue
            o_w, c_w = estimate_offset(e_ref[i0:i1], e_oth, fs, excl)
            ests.append(o_w - a)            # Fensterstart zurückrechnen
        if len(ests) == 2:
            drift = (ests[1] - ests[0]) * fps
            if abs(drift) > cfg["max_drift_frames"] or any(abs(e - off) * fps > 1.0 for e in ests):
                ok = False; note = f"Prüffenster uneinig ({ests[0]:.3f}s / {ests[1]:.3f}s)"
    elif ov1 - ov0 < 5.0:
        note = "keine Überlappung"
    else:
        note = f"Konfidenz {conf:.1f} unter {cfg['min_confidence']}"
    return PairSync(ref=str(ref_wav), other=str(other_wav), offset_s=round(off, 4),
                    offset_frames=int(round(off * fps)), confidence=round(conf, 2),
                    overlap_ref=[round(ov0, 2), round(ov1, 2)], drift_frames=round(drift, 2), ok=ok, note=note)


def sync_charge(charge, media: dict) -> dict:
    """Alle FX3×a7-Paare je Interview-Ordner; WAVs der ORIGINALE unter _intern/autocut/work/audio."""
    fps = media["format"]["fps"]
    cfg = charge.config["sync"]
    audio_dir = charge.work / "audio"
    paare = []
    for ordner, grp in media["ordner"].items():
        for ref_path in grp.get("ton", []):
            ref_wav = extract_audio_16k(ref_path, audio_dir, cfg["sr"])
            for oth_path in grp.get("kontext", []):
                oth_wav = extract_audio_16k(oth_path, audio_dir, cfg["sr"])
                p = sync_pair(ref_wav, oth_wav, fps, cfg)
                p.ref, p.other = ref_path, oth_path
                paare.append(p.to_dict())
                print(f"[{ordner}] {Path(ref_path).name} × {Path(oth_path).name}: "
                      f"{p.offset_frames:+d} Frames, Konf {p.confidence}, ok={p.ok} {p.note}")
    out = {"fps": fps, "paare": paare}
    charge.write_json("sync.json", out)
    return out


def a7_for_cut(sync: dict, ref_clip: str, in_s: float, out_s: float) -> dict | None:
    cands = [p for p in sync.get("paare", []) if p["ref"] == ref_clip and p["ok"]
             and p["overlap_ref"][0] <= in_s and out_s <= p["overlap_ref"][1]]
    return max(cands, key=lambda p: p["confidence"]) if cands else None
```

```python
"""Aus der geprüften Cutlist die konkreten Timeline-Items berechnen (frame-genau, relativ zum Timeline-Anfang)."""
from __future__ import annotations

from dataclasses import dataclass, field, asdict

from .cutlist import Cutlist
from .media import seconds_to_frames
from .sync import a7_for_cut

MARKER_COLORS = {"oton": "Blue", "vo": "Yellow", "bild": "Green", "grafik": "Purple", "warnung": "Red", "broll": "Yellow"}


@dataclass
class Item:
    track: str
    clip: str
    src_in_f: int
    src_out_f: int
    rec_in_f: int
    rec_out_f: int
    enabled: bool = True
    beat_nr: str = ""
    kind: str = "oton"
    video_only: bool = False


@dataclass
class MarkerSpec:
    frame: int
    name: str
    note: str
    color: str
    duration: int = 1


@dataclass
class BeatPos:
    nr: str
    typ: str
    rec_in_f: int
    rec_out_f: int
    clip: str | None = None
    person: str | None = None


@dataclass
class TimelinePlan:
    fps: float
    width: int
    height: int
    items: list[Item] = field(default_factory=list)
    markers: list[MarkerSpec] = field(default_factory=list)
    beats: list[BeatPos] = field(default_factory=list)
    total_frames: int = 0

    def to_dict(self) -> dict:
        return asdict(self)


def clamp_handles(words: list[dict], in_s: float, out_s: float, h_in_s: float, h_out_s: float,
                  speaker: str | None) -> tuple[float, float]:
    a, b = in_s - h_in_s, out_s + h_out_s
    for w in words:
        if speaker and w.get("speaker") == speaker:
            continue
        ws, we = float(w["start"]), float(w["end"])
        if we <= in_s and we > a:      # fremdes Wort ragt in den Vorlauf
            a = we
        if ws >= out_s and ws < b:     # fremdes Wort ragt in den Nachlauf
            b = ws
    return round(max(0.0, a), 3), round(b, 3)


def _speaker_at(words: list[dict], in_s: float, out_s: float) -> str | None:
    inside = [w.get("speaker") for w in words if in_s <= (float(w["start"]) + float(w["end"])) / 2 <= out_s]
    return max(set(inside), key=inside.count) if inside else None


def _note(b) -> str:
    parts = []
    if b.typ == "oton" and b.cuts:
        parts.append("O-Ton: " + " […] ".join(c.text for c in b.cuts if c.text))
    if b.text:
        parts.append("Text: " + b.text)
    if b.bild_hinweis:
        parts.append("Bild: " + b.bild_hinweis)
    if b.caption:
        parts.append("Caption: " + b.caption)
    if b.kommentar:
        parts.append("Kommentar: " + b.kommentar)
    return "\n".join(parts)


def build_timeline_plan(cl: Cutlist, media: dict, sync: dict, words_by_clip: dict, cfg: dict) -> TimelinePlan:
    fps = float(media["format"]["fps"])
    tp = TimelinePlan(fps=fps, width=media["format"]["width"], height=media["format"]["height"])
    h_in, h_out = cfg["handle_in_frames"] / fps, cfg["handle_out_frames"] / fps
    pos = 0
    for i, b in enumerate(cl.beats):
        beat_start = pos
        label = f"#{b.nr} {b.szene}"
        if b.typ == "oton":
            words = words_by_clip.get(b.clip, [])
            dur_f = media["clips"][b.clip]["original"]["nb_frames"]
            for c in b.cuts:
                spk = _speaker_at(words, c.in_s, c.out_s)
                a, z = clamp_handles(words, c.in_s, c.out_s, 0 if c.hart_in else h_in, 0 if c.hart_out else h_out, spk)
                s_in, s_out = seconds_to_frames(a, fps), min(seconds_to_frames(z, fps), dur_f)
                n = s_out - s_in
                tp.items.append(Item("V1", b.clip, s_in, s_out, pos, pos + n, True, b.nr, "oton"))
                tp.items.append(Item("A1", b.clip, s_in, s_out, pos, pos + n, True, b.nr, "oton"))
                pair = a7_for_cut(sync, b.clip, a, z)
                if pair:
                    off = int(pair["offset_frames"])
                    tp.items.append(Item("V2", pair["other"], s_in + off, s_out + off, pos, pos + n, True, b.nr, "oton"))
                    tp.items.append(Item("A2", pair["other"], s_in + off, s_out + off, pos, pos + n, False, b.nr, "oton"))
                else:
                    tp.markers.append(MarkerSpec(pos, f"V2 fehlt {label}", "Keine a7-Abdeckung für diesen Schnitt", MARKER_COLORS["warnung"]))
                pos += n
        else:
            pos += seconds_to_frames(float(b.platzhalter_s or 0.0), fps)
        tp.beats.append(BeatPos(b.nr, b.typ, beat_start, pos, b.clip, b.person))
        tp.markers.append(MarkerSpec(beat_start, label, _note(b), MARKER_COLORS.get(b.typ, "Blue")))
        if i < len(cl.beats) - 1:
            pause = b.pause_after_s if b.pause_after_s is not None else cl.pause_s
            pos += seconds_to_frames(float(pause), fps)
    tp.total_frames = pos if cl.beats and cl.beats[-1].typ != "oton" else (tp.items[-1].rec_out_f if tp.items else pos)
    tp.total_frames = max(tp.total_frames, pos)
    return tp
```

```python
# src/niro_autocut/resolve_api.py
"""DaVinci Resolve Scripting API — nur Anlegen neuer Bins/Timelines/Media-Pool-Einträge, nie Bestehendes ändern."""
from __future__ import annotations

import os
import sys
from pathlib import Path

from .charge import AutoCutError
from .timeline_model import Item, MarkerSpec, TimelinePlan

API = "/Library/Application Support/Blackmagic Design/DaVinci Resolve/Developer/Scripting"
LIB = "/Applications/DaVinci Resolve/DaVinci Resolve.app/Contents/Libraries/Fusion/fusionscript.so"
TRACK_INDEX = {"V1": 1, "V2": 2, "V3": 3, "A1": 1, "A2": 2}


def connect():
    os.environ.setdefault("RESOLVE_SCRIPT_API", API)
    os.environ.setdefault("RESOLVE_SCRIPT_LIB", LIB)
    mod = str(Path(API) / "Modules")
    if mod not in sys.path:
        sys.path.append(mod)
    try:
        import DaVinciResolveScript as dvr  # type: ignore
    except Exception as e:  # pragma: no cover
        raise AutoCutError(f"Resolve-Scripting-Modul nicht ladbar ({e}). Ist DaVinci Resolve Studio installiert?")
    r = dvr.scriptapp("Resolve")
    if not r:
        raise AutoCutError("DaVinci Resolve ist nicht erreichbar. Resolve starten, Projekt öffnen und in den "
                           "Einstellungen 'Externes Scripting: Lokal' erlauben.")
    return r


class ResolveSession:
    def __init__(self, resolve):
        self.resolve = resolve
        self.pm = resolve.GetProjectManager()
        self.project = self.pm.GetCurrentProject()
        if not self.project:
            raise AutoCutError("In Resolve ist kein Projekt geöffnet.")
        self.media_pool = self.project.GetMediaPool()

    # --- Media Pool ------------------------------------------------------
    def _walk(self, folder):
        yield folder
        for sub in folder.GetSubFolderList() or []:
            yield from self._walk(sub)

    def find_media_item(self, path: str):
        target = os.path.normpath(str(path))
        for folder in self._walk(self.media_pool.GetRootFolder()):
            for clip in folder.GetClipList() or []:
                try:
                    fp = clip.GetClipProperty("File Path")
                except Exception:
                    fp = None
                if fp and os.path.normpath(str(fp)) == target:
                    return clip
        return None

    def ensure_bin(self, parts: list[str]):
        folder = self.media_pool.GetRootFolder()
        for name in parts:
            existing = next((f for f in (folder.GetSubFolderList() or []) if f.GetName() == name), None)
            folder = existing or self.media_pool.AddSubFolder(folder, name)
            if not folder:
                raise AutoCutError(f"Bin '{name}' konnte nicht angelegt werden.")
        self.media_pool.SetCurrentFolder(folder)
        return folder

    def import_media(self, paths: list[str], folder) -> dict:
        self.media_pool.SetCurrentFolder(folder)
        out, missing = {}, []
        for p in paths:
            item = self.find_media_item(p)
            if item is None:
                missing.append(p)
            else:
                out[p] = item
        if missing:
            imported = self.media_pool.ImportMedia(missing) or []
            for it in imported:
                out[os.path.normpath(str(it.GetClipProperty("File Path")))] = it
            for p in missing:
                if p not in out:
                    it = self.find_media_item(p)
                    if it is None:
                        raise AutoCutError(f"Import in den Media Pool fehlgeschlagen: {p}")
                    out[p] = it
        return out

    def link_proxy(self, item, proxy_path: str | None) -> bool:
        if not proxy_path:
            return False
        try:
            return bool(item.LinkProxyMedia(str(proxy_path)))
        except Exception:
            return False

    # --- Timeline --------------------------------------------------------
    def create_timeline(self, name: str, fps: float, width: int, height: int, start_tc: str):
        existing = {self.project.GetTimelineByIndex(i).GetName() for i in range(1, self.project.GetTimelineCount() + 1)}
        if name in existing:
            raise AutoCutError(f"Timeline '{name}' existiert bereits.")
        t = self.media_pool.CreateEmptyTimeline(name)
        if not t:
            raise AutoCutError("Timeline konnte nicht angelegt werden.")
        t.SetSetting("useCustomSettings", "1")
        t.SetSetting("timelineFrameRate", str(fps).rstrip("0").rstrip("."))
        t.SetSetting("timelineResolutionWidth", str(width))
        t.SetSetting("timelineResolutionHeight", str(height))
        t.SetSetting("timelineOutputResolutionWidth", str(width))
        t.SetSetting("timelineOutputResolutionHeight", str(height))
        self.project.SetCurrentTimeline(t)
        return t

    def ensure_tracks(self, timeline, video: int, audio: int, names: dict) -> None:
        while timeline.GetTrackCount("video") < video:
            timeline.AddTrack("video")
        while timeline.GetTrackCount("audio") < audio:
            timeline.AddTrack("audio", "stereo")
        for key, name in names.items():
            kind = "video" if key.startswith("V") else "audio"
            timeline.SetTrackName(kind, TRACK_INDEX[key], name)

    def timeline_start_frame(self, timeline) -> int:
        return int(timeline.GetStartFrame())

    def append_items(self, timeline, items: list[Item], media_items: dict, start_frame: int) -> list:
        results = []
        for it in items:
            mi = media_items.get(it.clip) or media_items.get(os.path.normpath(it.clip))
            if mi is None:
                raise AutoCutError(f"Kein Media-Pool-Eintrag für {it.clip}")
            info = {"mediaPoolItem": mi, "startFrame": int(it.src_in_f), "endFrame": int(it.src_out_f) - 1,
                    "recordFrame": int(start_frame + it.rec_in_f), "trackIndex": TRACK_INDEX[it.track],
                    "mediaType": 1 if it.track.startswith("V") else 2}
            added = self.media_pool.AppendToTimeline([info]) or []
            if not added:
                raise AutoCutError(f"AppendToTimeline fehlgeschlagen für {Path(it.clip).name} auf {it.track} bei Frame {info['recordFrame']}")
            tl_item = added[0]
            if not it.enabled:
                tl_item.SetClipEnabled(False)
            results.append(tl_item)
        return results

    def add_markers(self, timeline, markers: list[MarkerSpec], start_frame: int) -> None:
        # frameId ist RELATIV zum Timeline-Start (siehe Recherche-Korrekturen) — start_frame wird NICHT addiert
        for m in markers:
            timeline.AddMarker(int(m.frame), m.color, m.name[:80], m.note[:2000], max(1, int(m.duration)), "")

    def export_timeline(self, timeline, path: str | Path, kind: str = "fcp7xml") -> Path:
        r = self.resolve
        kinds = {"fcp7xml": getattr(r, "EXPORT_FCP_7_XML", None), "otio": getattr(r, "EXPORT_OTIO", None),
                 "edl": getattr(r, "EXPORT_EDL", None)}
        k = kinds[kind]
        ok = timeline.Export(str(path), k, getattr(r, "EXPORT_NONE", None)) if kind == "edl" else timeline.Export(str(path), k)
        if not ok:
            raise AutoCutError(f"Export ({kind}) fehlgeschlagen: {path}")
        return Path(path)

    def read_timeline(self, timeline) -> dict:
        out = {"name": timeline.GetName(), "fps": timeline.GetSetting("timelineFrameRate"),
               "start_frame": timeline.GetStartFrame(), "tracks": {}, "markers": timeline.GetMarkers() or {}}
        for kind in ("video", "audio"):
            for i in range(1, timeline.GetTrackCount(kind) + 1):
                rows = []
                for it in timeline.GetItemListInTrack(kind, i) or []:
                    mpi = it.GetMediaPoolItem()
                    rows.append({"name": it.GetName(), "file": mpi.GetClipProperty("File Path") if mpi else None,
                                 "start": it.GetStart(), "end": it.GetEnd(), "src_in": it.GetSourceStartFrame(),
                                 "src_out": it.GetSourceEndFrame(), "enabled": it.GetClipEnabled()})
                out["tracks"][f"{kind[0].upper()}{i}"] = {"name": timeline.GetTrackName(kind, i), "items": rows}
        return out


def build_from_plan(session: ResolveSession, tp: TimelinePlan, media: dict, name: str, cfg: dict) -> dict:
    rcfg = cfg["resolve"]
    warnings: list[str] = []
    folder = session.ensure_bin([rcfg["bin_root"], name.replace("/", "-")[:60]])
    clips = sorted({it.clip for it in tp.items})
    media_items = session.import_media(clips, folder)
    for c in clips:
        proxy = (media["clips"].get(c) or {}).get("proxy_path")
        if proxy and not session.link_proxy(media_items[c], proxy):
            warnings.append(f"Proxy nicht verknüpft: {Path(c).name}")
    timeline = session.create_timeline(name, tp.fps, tp.width, tp.height, rcfg["start_timecode"])
    session.ensure_tracks(timeline, 3, 2, rcfg["track_names"])
    start = session.timeline_start_frame(timeline)
    order = {"V1": 0, "A1": 1, "V2": 2, "A2": 3, "V3": 4}
    items = sorted(tp.items, key=lambda it: (it.rec_in_f, order[it.track]))
    session.append_items(timeline, items, media_items, start)
    session.add_markers(timeline, tp.markers, start)
    return {"timeline": name, "items": len(items), "markers": len(tp.markers), "warnings": warnings, "start_frame": start}
```

```python
def discover_broll(index_records: list[dict]) -> list[dict]:
    roots: dict[Path, str | None] = {}
    for rec in index_records:
        p = Path(rec["path"])
        for anc in p.parents:
            if anc.name == "Sortiert":
                standort = anc.parent.name if anc.parent.name.lower().startswith("standort") else None
                roots[anc / "B-Roll"] = standort
                break
    clips: list[dict] = []
    for root, standort in sorted(roots.items()):
        if not root.is_dir():
            continue
        for f in sorted(root.rglob("*")):
            if f.is_file() and f.suffix.lower() in VIDEO_EXTS and "Proxy" not in f.parts:
                clips.append({"path": str(f), "ordner": f.parent.name if f.parent != root else "", "standort": standort})
    return clips


def scene_cuts(proxy: Path, threshold: float) -> list[float]:
    cmd = [_which("ffmpeg"), "-hide_banner", "-i", str(proxy), "-vf", f"select='gt(scene,{threshold})',showinfo",
           "-an", "-f", "null", "-"]
    r = subprocess.run(cmd, capture_output=True, text=True)
    return [float(m) for m in re.findall(r"pts_time:\s*([0-9.]+)", r.stderr)]


def frame_times(duration_s: float, cuts: list[float], cfg: dict) -> list[float]:
    bounds = [0.0] + sorted(c for c in cuts if 0.5 < c < duration_s - 0.5) + [duration_s]
    times: list[float] = []
    for a, b in zip(bounds, bounds[1:]):
        if b - a < 1.0:
            times.append((a + b) / 2); continue
        times += [a + 0.5, (a + b) / 2, b - 0.5]
        t = a + cfg["frame_interval_s"]
        while t < b - 0.5:
            times.append(t); t += cfg["frame_interval_s"]
    times = sorted(set(round(t, 2) for t in times if 0 <= t <= duration_s))
    if len(times) < cfg["frames_min"]:
        step = duration_s / (cfg["frames_min"] + 1)
        times = sorted(set(times + [round(step * (i + 1), 2) for i in range(cfg["frames_min"])]))
    if len(times) > cfg["frames_max"]:
        idx = [round(i * (len(times) - 1) / (cfg["frames_max"] - 1)) for i in range(cfg["frames_max"])]
        times = [times[i] for i in idx]
    return times


def extract_frames(proxy: Path, times: list[float], out_dir: Path, tile_px: int, portrait: bool) -> list[tuple[Path, float]]:
    out_dir.mkdir(parents=True, exist_ok=True)
    scale = f"scale=-2:{tile_px}" if portrait else f"scale={tile_px}:-2"
    frames = []
    for t in times:
        out = out_dir / f"{proxy.stem}_{int(round(t * 100)):07d}.jpg"
        if not out.exists():
            cmd = [_which("ffmpeg"), "-hide_banner", "-loglevel", "error", "-y", "-ss", f"{t:.3f}", "-i", str(proxy),
                   "-frames:v", "1", "-vf", scale, "-q:v", "4", str(out)]
            subprocess.run(cmd, capture_output=True, text=True)
        if out.exists():
            frames.append((out, t))
    return frames


def _font():
    for p in ("/System/Library/Fonts/Helvetica.ttc", "/Library/Fonts/Arial.ttf"):
        try:
            return ImageFont.truetype(p, 22)
        except Exception:
            continue
    return ImageFont.load_default()


def contact_sheets(frames: list[tuple[Path, float]], grid: tuple[int, int], out_prefix: Path) -> list[Path]:
    cols, rows = grid
    per = cols * rows
    sheets: list[Path] = []
    if not frames:
        return sheets
    w, h = Image.open(frames[0][0]).size
    font = _font()
    for n in range(0, len(frames), per):
        chunk = frames[n:n + per]
        r = (len(chunk) + cols - 1) // cols
        sheet = Image.new("RGB", (cols * w, r * h), (16, 16, 16))
        d = ImageDraw.Draw(sheet)
        for i, (p, t) in enumerate(chunk):
            im = Image.open(p).convert("RGB")
            if im.size != (w, h):
                im = im.resize((w, h))
            x, y = (i % cols) * w, (i // cols) * h
            sheet.paste(im, (x, y))
            label = f"{int(t // 60):02d}:{t % 60:04.1f}"
            d.rectangle([x, y, x + 96, y + 30], fill=(0, 0, 0))
            d.text((x + 6, y + 4), label, fill=(255, 255, 0), font=font)
        out = Path(f"{out_prefix}_{n // per + 1}.jpg")
        sheet.save(out, quality=82)
        sheets.append(out)
    return sheets


def _image_block(path: Path) -> dict:
    data = base64.standard_b64encode(path.read_bytes()).decode("ascii")
    return {"type": "image", "source": {"type": "base64", "media_type": "image/jpeg", "data": data}}


def describe_clip(client, sheets: list[Path], meta: dict, cfg: dict, system_prompt: str) -> dict:
    content = [_image_block(s) for s in sheets]
    content.append({"type": "text", "text": json.dumps({
        "datei": meta["name"], "ordner": meta.get("ordner"), "standort": meta.get("standort"),
        "dauer_s": meta["dauer_s"], "orientierung": meta["orientierung"], "szenenwechsel_s": meta.get("cuts", []),
        "kontaktbogen_hinweis": "Kacheln links oben → rechts unten, Zeitstempel mm:ss.s je Kachel"}, ensure_ascii=False)})
    for attempt in range(4):
        try:
            resp = client.messages.create(
                model=cfg["model"], max_tokens=cfg.get("max_tokens", 4000),
                system=[{"type": "text", "text": system_prompt, "cache_control": {"type": "ephemeral"}}],
                messages=[{"role": "user", "content": content}],
                output_config={"format": {"type": "json_schema", "schema": CLIP_SCHEMA}},
            )
            text = next(b.text for b in resp.content if b.type == "text")
            data = json.loads(text)
            data["_usage"] = {"input": resp.usage.input_tokens, "output": resp.usage.output_tokens,
                              "cache_read": getattr(resp.usage, "cache_read_input_tokens", 0)}
            return data
        except Exception as e:  # Ratelimit/Netz → Backoff
            if attempt == 3:
                raise AutoCutError(f"Claude-Anfrage für {meta['name']} fehlgeschlagen: {e}")
            time.sleep(2 ** attempt * 3)
    raise AutoCutError("unerreichbar")


def index_clip(charge, clip: dict, client, cfg: dict, system_prompt: str) -> dict:
    icfg = cfg["index"]
    path = Path(clip["path"])
    fp = fingerprint(path)
    cache_dir = charge.autocut / "broll_index"; cache_dir.mkdir(exist_ok=True)
    cached = cache_dir / f"{fp}.json"
    if cached.exists():
        return json.loads(cached.read_text(encoding="utf-8"))
    proxy = proxy_for(path) or path
    info = ffprobe(proxy)
    cuts = scene_cuts(proxy, icfg["scene_threshold"])
    times = frame_times(info.duration_s, cuts, icfg)
    frames = extract_frames(proxy, times, charge.work / "frames" / path.stem, icfg["tile_px"], is_portrait(info))
    sheets = contact_sheets(frames, tuple(icfg["grid"]), charge.work / "sheets" / path.stem)
    meta = {"name": path.name, "ordner": clip.get("ordner"), "standort": clip.get("standort"),
            "dauer_s": round(info.duration_s, 2), "orientierung": "9:16" if is_portrait(info) else "16:9", "cuts": [round(c, 2) for c in cuts]}
    data = describe_clip(client, sheets, meta, icfg, system_prompt)
    rec = {"path": str(path), "proxy": str(proxy), "fingerprint": fp, **meta, **data}
    cached.write_text(json.dumps(rec, ensure_ascii=False, indent=1), encoding="utf-8")
    return rec


def estimate_cost(n_clips: int) -> str:
    return f"~{n_clips} Clips × ~6.000 Token Eingabe + ~600 Token Ausgabe ≈ {n_clips * 0.045:.0f} € (Opus 5)"


def index_broll(charge, clips: list[dict], cfg: dict, limit: int | None = None, parallel: int = 4) -> dict:
    import anthropic
    from dotenv import load_dotenv
    load_dotenv(TOOL_ROOT / ".env")
    client = anthropic.Anthropic()
    system_prompt = (TOOL_ROOT / "prompts" / "index-clip.md").read_text(encoding="utf-8")
    todo = clips[:limit] if limit else clips
    results: dict[str, dict] = {}
    errors: list[str] = []
    with ThreadPoolExecutor(max_workers=parallel) as ex:
        futs = {ex.submit(index_clip, charge, c, client, cfg, system_prompt): c for c in todo}
        for i, fut in enumerate(as_completed(futs), 1):
            c = futs[fut]
            try:
                rec = fut.result(); results[rec["path"]] = rec
                print(f"[{i}/{len(todo)}] {Path(c['path']).name}: {rec.get('beschreibung_kurz', '')}")
            except AutoCutError as e:
                errors.append(str(e)); print(f"[{i}/{len(todo)}] FEHLER {Path(c['path']).name}: {e}")
    out = {"clips": [results[k] for k in sorted(results)], "fehler": errors, "anzahl": len(results)}
    charge.write_json("broll_index.json", out)
    return out
```
